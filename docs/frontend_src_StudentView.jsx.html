import { useState, useEffect } from "react";
import Table from "react-bootstrap/Table";
import { Form } from "react-bootstrap";
import Button from "react-bootstrap/Button";
import Dropdown from "react-bootstrap/Dropdown";

/**
 * StudentView component for practicing foreign language grammar.
 * @component
 * @param {Object} props - The component props.
 * @param {Array} props.words - The list of words.
 * @param {Array} props.tags - The list of tags.
 * @param {Function} props.setRole - Function to update the role state.
 * @return {JSX.Element} The rendered component.
 */
function StudentView({ words, tags, setRole }) {
  const [userAnswers, setUserAnswers] = useState([]);
  const [rightAnswers, setRightAnswers] = useState([]);
  const [userCorrectAnswers, setUserCorrectAnswers] = useState([]);
  const [activeTag, setActiveTag] = useState([]);
  const [filteredRightAnswers, setFilteredRightAnswers] = useState([]);
  const [isChecking, setChecking] = useState(false);
  const [feedbackMessage, setFeedbackMessage] = useState("");
  const [isTableVisible, setIsTableVisible] = useState(false);
  const [isTagButtonsVisible, setIsTagButtonsVisible] = useState(false);
  const [isEnglishSelected, setIsEnglishSelected] = useState(false);
  const [showAll, setShowAll] = useState(false);

  /**
   * Checks the user's answers for correctness, calculates the points
   * and provides a feedback message.
   * @function
   */
  const checkAnswers = () => {
    let points = 0;
    let correctAnswers = [];
    filteredRightAnswers.forEach((rightAnswer, index) => {
      if (userAnswers[rightAnswer.id - 1]) {
        if (
          rightAnswer.word.trim().toLowerCase() ===
          userAnswers[rightAnswer.id - 1].trim().toLowerCase()
        ) {
          points++;
          correctAnswers.push(rightAnswer.id);
        }
      }
    });
    setUserCorrectAnswers(correctAnswers);
    setFeedbackMessage(
      <p
        style={{
          marginTop: 10,
          color: points === filteredRightAnswers.length ? "green" : "red",
        }}
      >
        You got: {points}/{filteredRightAnswers.length} points!
      </p>
    );
    setChecking(true);
  };

  /**
   * Handles the change event for the answer input fields.
   * @param {Object} event - The change event.
   * @param {number} id - The ID of the word row.
   */
  const handleInputChange = (event, id) => {
    const newAnswers = [...userAnswers];
    newAnswers[id - 1] = event.target.value;
    setUserAnswers(newAnswers);
  };

  /**
   * Toggles the language the user is practicing.
   * Updates right answers state according to the selected language.
   * @param {string} language - The selected language ("english" or "finnish").
   */
  const toggleLanguage = (language) => {
    if (language === "english") {
      setIsEnglishSelected(true);
      let englishRightAnswers = words.map((pair) => {
        return { id: pair.id, word: pair.english, tag: pair.tag };
      });
      setRightAnswers(englishRightAnswers);
    } else {
      setIsEnglishSelected(false);
      let finnishRightAnswers = words.map((pair) => {
        return { id: pair.id, word: pair.finnish, tag: pair.tag };
      });
      setRightAnswers(finnishRightAnswers);
    }
    setIsTagButtonsVisible(true);
    setIsTableVisible(false);
  };

  /**
   * Toggles the tag for filtering the words.
   * @param {number} tag - The selected tag ID.
   */
  const toggleTag = (tag) => {
    setActiveTag(tag);
    setIsTableVisible(true);
    setUserAnswers([]);
    if (tag === 0) {
      setShowAll(true);
      setFilteredRightAnswers(rightAnswers);
    } else {
      let newRightAnswers = rightAnswers;
      setFilteredRightAnswers(
        newRightAnswers.filter((word) => Number(word.tag) === tag)
      );
      setShowAll(false);
    }
  };

  return (
    <>
      <Dropdown style={{ textAlign: "right" }}>
        <Dropdown.Toggle variant="secondary">Menu</Dropdown.Toggle>
        <Dropdown.Menu>
          <Dropdown.Item onClick={() => setRole("")}>Switch role</Dropdown.Item>
        </Dropdown.Menu>
      </Dropdown>
      <h2 style={{ margin: 10 }}>Student</h2>
      <p>Select which language you are writing in.</p>
      <Button
        variant={isEnglishSelected ? "primary" : "outline-primary"}
        className="m-2"
        onClick={() => toggleLanguage("english")}
      >
        English
      </Button>
      <Button
        variant={
          isEnglishSelected || !isTagButtonsVisible
            ? "outline-primary"
            : "primary"
        }
        className="m-2"
        onClick={() => toggleLanguage("finnish")}
      >
        Finnish
      </Button>
      {isTagButtonsVisible && (
        <div id="tagButtons">
          <p style={{ marginTop: 10 }}>
            Select the category you want, or select all.
          </p>
          {tags.map((tag) => {
            return (
              <Button
                key={tag.id}
                variant={
                  activeTag === tag.id ? "secondary" : "outline-secondary"
                }
                className="m-1"
                onClick={() => toggleTag(tag.id)}
              >
                {tag.name}
              </Button>
            );
          })}
          <Button
            variant={activeTag === 0 ? "secondary" : "outline-secondary"}
            className="m-1"
            onClick={() => toggleTag(0)}
          >
            All
          </Button>
        </div>
      )}
      {isTableVisible && (
        <Table
          className="custom-table"
          id="studentTable"
          striped
          bordered
          hover
          size="sm"
          responsive
        >
          <thead>
            <tr>
              {isEnglishSelected ? (
                <>
                  <th>Finnish</th>
                  <th>English</th>
                </>
              ) : (
                <>
                  <th>English</th>
                  <th>Finnish</th>
                </>
              )}
            </tr>
          </thead>
          <tbody>
            {words.map((pair, index) => {
              if (Number(pair.tag) === activeTag || showAll) {
                return (
                  <tr key={pair.id}>
                    <td>
                      <p
                        style={{
                          color: isChecking
                            ? userCorrectAnswers.includes(pair.id)
                              ? "green"
                              : "red"
                            : "black",
                          margin: 0,
                          marginTop: 6,
                          fontWeight: 500,
                        }}
                      >
                        {isEnglishSelected ? pair.finnish : pair.english}
                      </p>
                    </td>
                    <td>
                      <Form.Control
                        style={{
                          width: "100%",
                          color: isChecking
                            ? userCorrectAnswers.includes(pair.id)
                              ? "green"
                              : "red"
                            : "black",
                          borderRadius: 0,
                        }}
                        type="text"
                        value={userAnswers[pair.id - 1] || ""}
                        onChange={(e) => handleInputChange(e, pair.id)}
                      />
                    </td>
                  </tr>
                );
              }
            })}
          </tbody>
          <tfoot>
            <tr>
              <td colSpan="1">
                <Button
                  style={{ width: 70, marginTop: 3, marginBottom: 2 }}
                  variant="success"
                  onClick={checkAnswers}
                  onBlur={() => setChecking(false)}
                >
                  Check
                </Button>
              </td>
              <td style={{ fontWeight: 500 }}>
                {isChecking ? feedbackMessage : null}
              </td>
            </tr>
          </tfoot>
        </Table>
      )}
    </>
  );
}
export default StudentView;
